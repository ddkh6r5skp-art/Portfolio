async function fetchPortfolioData(userPasscode) { try { // Passes the passcode dynamically to Google Apps Script const res = await fetch(`${API_URL}?passcode=${encodeURIComponent(userPasscode)}`); const data = await res.json(); // If server rejected the passcode if (data.error) { document.getElementById("error-msg").innerText = "Invalid passcode. Access denied by server."; document.getElementById("error-msg").classList.remove("hidden"); sessionStorage.removeItem("authenticated"); return; } // Passcode verified! Store session and display data sessionStorage.setItem("authenticated", userPasscode); showDashboardUI(); // Populate KPIs document.getElementById("total-invested").innerText = formatCurrency(data.totalInvested); document.getElementById("net-worth").innerText = formatCurrency(data.netWorth); const profitElem = document.getElementById("net-profit"); profitElem.innerText = formatCurrency(data.netProfit); profitElem.className = data.netProfit >= 0 ? "text-2xl font-bold text-emerald-400 mt-1" : "text-2xl font-bold text-red-400 mt-1"; document.getElementById("return-pct").innerText = (data.returnPct * 100).toFixed(2) + "%"; document.getElementById("last-updated").innerText = "Last synced: " + new Date(data.lastUpdated).toLocaleTimeString(); renderChart(data.netWorth, data.returnPct); } catch (err) { console.error("Failed to load portfolio data", err); } }
